Local IoT + RAG Logistics Guide (Chroma)

Prototype: Mock IoT telemetry → Detention detection → RAG retrieval (Chroma) → Agent recommendations

What this guide builds

  • Mock telemetry generator (Python asyncio) that simulates trucks arriving, waiting, and loading.
  • Detention detector — a stateful stream processor that emits detention_started events.
  • RAG ingestion — chunk contracts/SOPs and store embeddings in Chroma.
  • RAG agent — retrieve relevant clauses and call an LLM to produce structured JSON recommendations.
  • All components run locally with an in‑process queue (no external broker required).

Step 1 — Mock telemetry generator

Save as src/mock_generator.py.

# src/mock_generator.py
import asyncio
import random
import time
import uuid

async def truck_simulator(out_queue: asyncio.Queue, truck_id: str, stop_id: str):
    arrival_ts = time.time()
    await out_queue.put({
        "event_type": "arrival",
        "truck_id": truck_id,
        "stop_id": stop_id,
        "timestamp": arrival_ts
    })
    # simulate variable delay before check-in
    await asyncio.sleep(random.uniform(1, 5))
    await out_queue.put({
        "event_type": "checkin",
        "truck_id": truck_id,
        "stop_id": stop_id,
        "timestamp": time.time()
    })
    # simulate waiting for dock assignment (random)
    wait_for_dock = random.uniform(30, 300)  # seconds
    start = time.time()
    while time.time() - start < wait_for_dock:
        await asyncio.sleep(random.uniform(5, 20))
        await out_queue.put({
            "event_type": "status",
            "truck_id": truck_id,
            "stop_id": stop_id,
            "status": "waiting_for_dock",
            "timestamp": time.time()
        })
    # simulate dock assigned and loading start
    await out_queue.put({
        "event_type": "dock_assigned",
        "truck_id": truck_id,
        "stop_id": stop_id,
        "dock_id": f"dock-{random.randint(1,10)}",
        "timestamp": time.time()
    })
    await asyncio.sleep(random.uniform(2, 6))
    await out_queue.put({
        "event_type": "loading_start",
        "truck_id": truck_id,
        "stop_id": stop_id,
        "timestamp": time.time()
    })

async def start_generators(out_queue: asyncio.Queue, n=5):
    tasks = []
    for i in range(n):
        truck_id = f"TRK-{str(uuid.uuid4())[:8]}"
        stop_id = "WH-12"
        tasks.append(asyncio.create_task(truck_simulator(out_queue, truck_id, stop_id)))
        await asyncio.sleep(0.5)
    await asyncio.gather(*tasks)

Step 2 — Detention detector

Save as src/detector.py. This keeps a per‑truck state machine and emits detention events when free time is exceeded.

# src/detector.py
import asyncio
import time
from typing import Dict

class DetentionDetector:
    def __init__(self, free_time_seconds=120):
        self.free_time = free_time_seconds
        self.state: Dict[str, dict] = {}  # truck_id -> {arrival_ts, checkin_ts, dock_assigned, loading_started}

    async def run(self, in_queue: asyncio.Queue, out_queue: asyncio.Queue):
        while True:
            ev = await in_queue.get()
            t = ev.get("truck_id")
            now = ev.get("timestamp", time.time())
            if t not in self.state:
                self.state[t] = {}
            s = self.state[t]

            if ev["event_type"] == "arrival":
                s["arrival_ts"] = now
                s["stop_id"] = ev.get("stop_id")
            elif ev["event_type"] == "checkin":
                s["checkin_ts"] = now
            elif ev["event_type"] == "dock_assigned":
                s["dock_assigned"] = now
            elif ev["event_type"] == "loading_start":
                s["loading_started"] = now

            # compute detention
            arrival = s.get("arrival_ts")
            loading = s.get("loading_started")
            if arrival and not loading:
                waited = time.time() - arrival
                if waited > self.free_time:
                    # emit detention event
                    detention_event = {
                        "event_type": "detention_started",
                        "truck_id": t,
                        "stop_id": s.get("stop_id"),
                        "arrival_ts": arrival,
                        "minutes_waiting": int(waited / 60),
                        "reason": "no_loading_started",
                        "detected_at": time.time()
                    }
                    await out_queue.put(detention_event)
            in_queue.task_done()

Step 3 — RAG ingestion into Chroma

Save as src/rag_ingest.py. This chunks contract text, embeds with sentence-transformers, and writes to Chroma.

# src/rag_ingest.py
from sentence_transformers import SentenceTransformer
import chromadb
from chromadb.config import Settings
import os
import glob

MODEL_NAME = "all-MiniLM-L6-v2"

def chunk_text(text, max_chars=800):
    paragraphs = text.split("\n\n")
    chunks = []
    current = ""
    for p in paragraphs:
        if len(current) + len(p) > max_chars:
            if current:
                chunks.append(current.strip())
            current = p
        else:
            current += "\n\n" + p
    if current:
        chunks.append(current.strip())
    return chunks

def ingest_contracts(data_dir="data/contracts", chroma_dir=".chromadb"):
    model = SentenceTransformer(MODEL_NAME)
    client = chromadb.Client(Settings(chroma_db_impl="duckdb+parquet", persist_directory=chroma_dir))
    collection = client.get_or_create_collection(name="logistics_docs")
    files = glob.glob(os.path.join(data_dir, "*.txt"))
    for f in files:
        with open(f, "r", encoding="utf-8") as fh:
            text = fh.read()
        chunks = chunk_text(text)
        ids = []
        texts = []
        metadatas = []
        for i, c in enumerate(chunks):
            ids.append(f"{os.path.basename(f)}-chunk-{i}")
            texts.append(c)
            metadatas.append({"source": os.path.basename(f), "section_index": i})
        embs = model.encode(texts, show_progress_bar=False).tolist()
        collection.add(ids=ids, documents=texts, metadatas=metadatas, embeddings=embs)
    client.persist()
    print("Ingest complete.")

Step 4 — RAG agent (retrieval + LLM)

Save as src/rag_agent.py. This retrieves top chunks and calls an LLM to produce structured JSON output.

# src/rag_agent.py
import chromadb
from chromadb.config import Settings
from sentence_transformers import SentenceTransformer
import os
import openai
import json

MODEL_NAME = "all-MiniLM-L6-v2"

class RAGAgent:
    def __init__(self, chroma_dir=".chromadb", openai_api_key=None):
        self.client = chromadb.Client(Settings(chroma_db_impl="duckdb+parquet", persist_directory=chroma_dir))
        self.collection = self.client.get_collection("logistics_docs")
        self.embed_model = SentenceTransformer(MODEL_NAME)
        if openai_api_key:
            openai.api_key = openai_api_key

    def retrieve(self, query_text, top_k=4):
        q_emb = self.embed_model.encode([query_text])[0].tolist()
        results = self.collection.query(query_embeddings=[q_emb], n_results=top_k)
        docs = []
        for doc, meta, score in zip(results["documents"][0], results["metadatas"][0], results["distances"][0]):
            docs.append({"text": doc, "metadata": meta, "score": score})
        return docs

    def build_prompt(self, event, docs):
        ctx = "\n\n---\n\n".join([d["text"] for d in docs])
        prompt = f"""
You are a logistics operations assistant. Use ONLY the context below to answer the questions.

Context:
{ctx}

Event:
{json.dumps(event, indent=2)}

Questions:
1) Is detention billable now? If so, why and what clause supports it.
2) Estimate detention cost so far (use a simple rule: $50 per hour).
3) Who should be notified and recommended actions.
Return a JSON object with keys: detention_billable (bool), reason, estimated_cost_usd (number), notifications (list), actions (list).
"""
        return prompt

    def call_llm(self, prompt):
        # Default: OpenAI completion. Swap for local LLM if desired.
        resp = openai.ChatCompletion.create(
            model="gpt-4o-mini",
            messages=[{"role":"system","content":"You are a JSON-output assistant."},
                      {"role":"user","content":prompt}],
            temperature=0
        )
        return resp["choices"][0]["message"]["content"]

    def analyze_event(self, event):
        q = f"detention for truck {event['truck_id']} at {event['stop_id']} waiting {event['minutes_waiting']} minutes"
        docs = self.retrieve(q)
        prompt = self.build_prompt(event, docs)
        llm_out = self.call_llm(prompt)
        try:
            parsed = json.loads(llm_out)
        except Exception:
            parsed = {"raw": llm_out}
        return parsed

Replace the model name with your available model or adapt call_llm to a local LLM endpoint.

Step 5 — Main orchestration

Save as src/main.py. This wires generator → detector → RAG agent and prints results.

# src/main.py
import asyncio
from mock_generator import start_generators
from detector import DetentionDetector
from rag_ingest import ingest_contracts
from rag_agent import RAGAgent
import os
import dotenv

dotenv.load_dotenv()

async def main():
    # ensure contracts ingested
    ingest_contracts(data_dir="data/contracts", chroma_dir=".chromadb")

    telemetry_q = asyncio.Queue()
    detention_q = asyncio.Queue()

    detector = DetentionDetector(free_time_seconds=120)
    rag = RAGAgent(chroma_dir=".chromadb", openai_api_key=os.getenv("OPENAI_API_KEY"))

    # start detector loop
    detector_task = asyncio.create_task(detector.run(telemetry_q, detention_q))
    # start generators
    gen_task = asyncio.create_task(start_generators(telemetry_q, n=6))

    async def rag_consumer():
        while True:
            ev = await detention_q.get()
            print("RAG processing event:", ev)
            result = rag.analyze_event(ev)
            print("RAG result:", result)
            detention_q.task_done()

    rag_task = asyncio.create_task(rag_consumer())

    await gen_task
    await telemetry_q.join()
    await asyncio.sleep(2)
    detector_task.cancel()
    rag_task.cancel()

if __name__ == "__main__":
    asyncio.run(main())

Troubleshooting & improvements

  • Chroma persistence: If you re-run ingestion, either remove .chromadb or use a new collection name.
  • Embedding model: Use a larger model for better retrieval quality if needed.
  • Parse numeric rates: Improve cost estimation by parsing numeric rates from contract chunks instead of fixed $50/hr.
  • Human approval UI: Add a small FastAPI + React UI to approve billing lines before sending to billing.
  • Local LLM: Replace OpenAI calls with a local LLM endpoint for offline use.
